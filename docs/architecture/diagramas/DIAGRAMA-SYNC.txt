╚══════════════════════════════════════════════════════════════════════════════════════════════╝

  ConflictResolutionStrategy = {
  
    LOCAL_WINS:    "Gana siempre el registro local (más reciente en dispositivo)",
    REMOTE_WINS:   "Gana siempre el registro remoto (más reciente en servidor)",  
    MERGE:         "Fusiona campos: local + remote → merged",
    MANUAL:        "Requiere resolución manual del usuario",
    TIMESTAMP:     "Gana el registro con timestamp más reciente",
    FIELD_PRIORITY: {
      barcode: "remote",
      name: "local", 
      price: "remote"
    }
  }

  applyResolution(conflict, strategy) {
    switch(strategy) {
      case 'LOCAL_WINS':    return conflict.localVersion;
      case 'REMOTE_WINS':   return conflict.remoteVersion;
      case 'MERGE':         return mergeRecords(conflict);
      case 'MANUAL':        return null; // User decides
      case 'TIMESTAMP':     return newer(conflict.localVersion, conflict.remoteVersion);
    }
  }
